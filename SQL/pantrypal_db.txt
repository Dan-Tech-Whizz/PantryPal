-- ---------- Tables ----------
CREATE TABLE USERS (
    user_id                    INT PRIMARY KEY AUTO_INCREMENT,
    name                       VARCHAR(100) NOT NULL,
    email                      VARCHAR(255) NOT NULL UNIQUE,
    password_hash              VARCHAR(255) NOT NULL,
    notifications_enabled      BOOLEAN NOT NULL DEFAULT TRUE,
    allow_purchase_suggestions BOOLEAN NOT NULL DEFAULT FALSE
) ENGINE=InnoDB;

CREATE TABLE CATEGORY (
    category_id     INT PRIMARY KEY AUTO_INCREMENT,
    category_name   VARCHAR(50) NOT NULL UNIQUE,
    shelf_life_days INT NOT NULL
) ENGINE=InnoDB;

CREATE TABLE INGREDIENT (
    ingredient_id   INT PRIMARY KEY AUTO_INCREMENT,
    ingredient_name VARCHAR(100) NOT NULL UNIQUE,
    category_id     INT NOT NULL,
    FOREIGN KEY (category_id) REFERENCES CATEGORY(category_id)
) ENGINE=InnoDB;

CREATE TABLE RECEIPT (
    receipt_id INT PRIMARY KEY AUTO_INCREMENT,
    user_id    INT NOT NULL,
    scanned_at DATETIME NOT NULL,
    image_path VARCHAR(255),
    FOREIGN KEY (user_id) REFERENCES USERS(user_id)
) ENGINE=InnoDB;

CREATE TABLE PANTRY_ITEM (
    item_id       INT PRIMARY KEY AUTO_INCREMENT,
    user_id       INT NOT NULL,
    ingredient_id INT NOT NULL,
    receipt_id    INT NULL,                       
    purchase_date DATE NOT NULL,
    expiry_date   DATE NOT NULL,
    quantity      INT NOT NULL DEFAULT 1,
    status        VARCHAR(10) NOT NULL DEFAULT 'ACTIVE',  -- ACTIVE, CONSUMED, EXPIRED, DISCARDED
    status_date   DATE,
    FOREIGN KEY (user_id)       REFERENCES USERS(user_id),
    FOREIGN KEY (ingredient_id) REFERENCES INGREDIENT(ingredient_id),
    FOREIGN KEY (receipt_id)    REFERENCES RECEIPT(receipt_id)
) ENGINE=InnoDB;

CREATE TABLE SHOPPING_LIST (
    list_id      INT PRIMARY KEY AUTO_INCREMENT,
    user_id      INT NOT NULL,
    created_date DATE NOT NULL,
    FOREIGN KEY (user_id) REFERENCES USERS(user_id)
) ENGINE=InnoDB;

CREATE TABLE SHOPPING_LIST_ITEM (
    list_item_id  INT PRIMARY KEY AUTO_INCREMENT,
    list_id       INT NOT NULL,
    ingredient_id INT NOT NULL,
    quantity      INT NOT NULL DEFAULT 1,
    is_purchased  BOOLEAN NOT NULL DEFAULT FALSE,
    added_date    DATE NOT NULL,
    UNIQUE (list_id, ingredient_id),             
    FOREIGN KEY (list_id)       REFERENCES SHOPPING_LIST(list_id),
    FOREIGN KEY (ingredient_id) REFERENCES INGREDIENT(ingredient_id)
) ENGINE=InnoDB;

CREATE TABLE RECIPE (
    recipe_id    INT PRIMARY KEY AUTO_INCREMENT,
    title        VARCHAR(150) NOT NULL,
    instructions TEXT,
    created_by   INT NULL,                       
    FOREIGN KEY (created_by) REFERENCES USERS(user_id)
) ENGINE=InnoDB;

CREATE TABLE RECIPE_INGREDIENT (
    recipe_id       INT NOT NULL,
    ingredient_id   INT NOT NULL,
    quantity_needed VARCHAR(50),
    PRIMARY KEY (recipe_id, ingredient_id),
    FOREIGN KEY (recipe_id)     REFERENCES RECIPE(recipe_id),
	FOREIGN KEY (ingredient_id) REFERENCES INGREDIENT(ingredient_id)
) ENGINE=InnoDB;

CREATE TABLE FAVOURITE_RECIPE (
    user_id    
    INT NOT NULL,
    recipe_id  INT NOT NULL,
    saved_date DATE NOT NULL,
    PRIMARY KEY (user_id, recipe_id),
    FOREIGN KEY (user_id)   REFERENCES USERS(user_id),
    FOREIGN KEY (recipe_id) REFERENCES RECIPE(recipe_id)
) ENGINE=InnoDB;

SELECT u.name, i.ingredient_name, c.category_name, p.quantity, p.expiry_date
FROM PANTRY_ITEM p
JOIN USERS u ON u.user_id = p.user_id
JOIN INGREDIENT i ON i.ingredient_id = p.ingredient_id
JOIN CATEGORY c ON c.category_id = i.category_id
ORDER BY p.item_id;
